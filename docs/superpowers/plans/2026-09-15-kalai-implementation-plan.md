# Kalai Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Kalai, a dark, earthy, completely free iOS health tracker with 1:1 functional parity to MyFitnessPal, built on a robust, bug-free SwiftData architecture.

**Architecture:** A completely fresh SwiftUI application designed using MVVM and SwiftData. Data ingestion will pull from extracted, cleaned JSON datasets derived from the original application's decrompiled assets. The UI will strictly follow a dark, earthy aesthetic via a centralized theme engine.

**Tech Stack:** Swift, SwiftUI, SwiftData, XcodeGen, Combine, VisionKit, CoreML (for OCR).

**Spec:** `/Users/dhruv/docs/superpowers/specs/2026-09-15-kalai-architecture-design.md`

## Global Constraints
- Minimum iOS 17.0.
- Do NOT use `UIKit` where SwiftUI is sufficient, to ensure maximum platform fluidity.
- App MUST compile with `DEVELOPER_DIR=/Applications/Xcode.app/Contents/Developer` and `CODE_SIGNING_ALLOWED=NO`.
- STRICTLY forbid pure black (`#000000`) or pure white (`#FFFFFF`) anywhere in the UI; all styling must go through `KalaiTheme`.
- TDD applies. Write failing tests before implementing logic.

```swift
import SwiftUI

enum KalaiTheme {
    struct Colors {
        let background = Color(red: 24/255, green: 26/255, blue: 23/255) // Dark Charcoal (#1C1E1B)
        let surface = Color(red: 63/255, green: 75/255, blue: 59/255)   // Muted Olive (#3F4B3B)
        let accent = Color(red: 194/255, green: 168/255, blue: 149/255) // Ochre (#C2A895)
        let text = Color(red: 240/255, green: 234/255, blue: 214/255)   // Soft Cream (#F0EAD6)
    }
    
    static let colors = Colors()
    
    static let spring = Animation.spring(response: 0.5, dampingFraction: 0.7)
}

extension View {
    func kalaiBackground() -> some View {
        self.background(KalaiTheme.colors.background.ignoresSafeArea())
    }
}
```

Create `Kalai/Sources/Models/UserProfile.swift`.

```swift
import Foundation
import SwiftData

@Model
class UserProfile {
    var name: String
    var weightKg: Double
    var heightCm: Double
    var age: Int
    var isMale: Bool
    var dailyCalorieGoal: Int = 2000
    
    init(name: String, weightKg: Double, heightCm: Double, age: Int, isMale: Bool) {
        self.name = name
        self.weightKg = weightKg
        self.heightCm = heightCm
        self.age = age
        self.isMale = isMale
    }
    
    func calculateBMR() -> Double {
        // Mifflin-St Jeor Equation
        if isMale {
            return (10 * weightKg) + (6.25 * heightCm) - (5 * Double(age)) + 5
        } else {
            return (10 * weightKg) + (6.25 * heightCm) - (5 * Double(age)) - 161
        }
    }
}
```

```swift
import Foundation
import SwiftData

@Model
class DailyDiary {
    var date: Date
    @Relationship(deleteRule: .cascade) var foods: [FoodRecord]
    
    init(date: Date = .now, foods: [FoodRecord] = []) {
        self.date = date
        self.foods = foods
    }
    
    var totalCalories: Double {
        foods.reduce(0) { $0 + $1.calories }
    }
}
```

```swift
struct CatalogFood: Codable, Identifiable {
```

```swift
import Foundation

enum DataIngestor {

            return []
        }
        return foods
    }
}
```

```swift
import SwiftUI

struct MacroRingView: View {
    let total: Double
    let consumed: Double
    
    var percentage: Double {
        min(consumed / total, 1.0)
    }
    
    var body: some View {
        ZStack {
            Circle()
                .stroke(KalaiTheme.colors.surface, lineWidth: 20)
            
            Circle()
                .trim(from: 0, to: percentage)
                .stroke(KalaiTheme.colors.accent, style: StrokeStyle(lineWidth: 20, lineCap: .round))
                .rotationEffect(.degrees(-90))
                .animation(.spring(), value: percentage)
        }
        .frame(width: 150, height: 150)
    }
}
```
